   292→export const CREATE_TRANSACTION_FN = `// ============================================================
   293→// Edge Function: create-transaction
   294→// مسیر فایل در پروژه: supabase/functions/create-transaction/index.ts
   295→// دیپلوی: supabase functions deploy create-transaction --no-verify-jwt
   296→// ============================================================
   297→// این تابع تراکنش مالی را با اعتبارسنجی اتمیک ثبت می‌کند.
   298→// کاربران مستقیماً به جدول transactions INSERT نمی‌کنند — همگی از این تابع می‌گذرند.
   299→
   300→import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
   301→
   302→const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
   303→const SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";
   304→
   305→const TRANSACTION_EFFECT: Record<string, { receivable: number; payable: number }> = {
   306→  OPENING_RECEIVABLE: { receivable: 1, payable: 0 },
   307→  OPENING_PAYABLE: { receivable: 0, payable: 1 },
   308→  RECEIVABLE_CHARGE: { receivable: 1, payable: 0 },
   309→  PAYABLE_CHARGE: { receivable: 0, payable: 1 },
   310→  RECEIPT_ALLOCATION: { receivable: -1, payable: 0 },
   311→  PAYMENT_ALLOCATION: { receivable: 0, payable: -1 },
   312→};
   313→
   314→const corsHeaders = {
   315→  "Access-Control-Allow-Origin": "*",
   316→  "Access-Control-Allow-Headers":
   317→    "authorization, x-client-info, apikey, content-type",
   318→};
   319→
   320→function json(data: unknown, status = 200) {
   321→  return new Response(JSON.stringify(data), {
   322→    status,
   323→    headers: { ...corsHeaders, "Content-Type": "application/json" },
   324→  });
   325→}
   326→
   327→// استخراج شناسه کاربر از JWT
   328→// (Supabase هنگام درخواست از سمت کلاینت با verify_jwt=true خودکار تأیید می‌کند،
   329→//  ولی چون این تابع را با --no-verify-jwt دیپلوی می‌کنیم، خودمان payload را استخراج می‌کنیم)
   330→function extractUser(jwt: string): { sub: string } | null {
   331→  try {
   332→    const parts = jwt.split(".");
   333→    if (parts.length !== 3) return null;
   334→    const payload = JSON.parse(atob(parts[1]));
   335→    if (!payload.sub) return null;
   336→    if (payload.exp && payload.exp < Date.now() / 1000) return null;
   337→    return { sub: payload.sub };
   338→  } catch {
   339→    return null;
   340→  }
   341→}
   342→
   343→async function computeOutstanding(
   344→  supabase: ReturnType<typeof createClient>,
   345→  contactId: string,
   346→  currency: string
   347→): Promise<{ receivable: number; payable: number }> {
   348→  const { data, error } = await supabase
   349→    .from("transactions")
   350→    .select("transaction_type, amount")
   351→    .eq("contact_id", contactId)
   352→    .eq("status", "POSTED")
   353→    .eq("currency", currency);
   354→
   355→  if (error) throw error;
   356→
   357→  let receivable = 0;
   358→  let payable = 0;
   359→  for (const t of data ?? []) {
   360→    const effect = TRANSACTION_EFFECT[t.transaction_type];
   361→    if (!effect) continue;
   362→    receivable += effect.receivable * t.amount;
   363→    payable += effect.payable * t.amount;
   364→  }
   365→  return { receivable, payable };
   366→}
   367→
   368→async function logAudit(
   369→  supabase: ReturnType<typeof createClient>,
   370→  userId: string,
   371→  result: "success" | "failure",
   372→  details: string,
   373→  entityId: string
   374→) {
   375→  await supabase.from("audit_logs").insert({
   376→    actor_id: userId,
   377→    action: "CREATE_TXN",
   378→    entity_type: "Transaction",
   379→    entity_id: entityId,
   380→    result,
   381→    details,
   382→  });
   383→}
   384→
   385→Deno.serve(async (req: Request) => {
   386→  if (req.method === "OPTIONS") {
   387→    return new Response("ok", { headers: corsHeaders });
   388→  }
   389→  if (req.method !== "POST") {
   390→    return json({ error: "Method not allowed" }, 405);
   391→  }
   392→
   393→  const authHeader = req.headers.get("Authorization");
   394→  if (!authHeader || !authHeader.startsWith("Bearer ")) {
   395→    return json({ error: "Unauthorized" }, 401);
   396→  }
   397→  const jwt = authHeader.replace("Bearer ", "");
   398→  const user = extractUser(jwt);
   399→  if (!user) {
   400→    return json({ error: "Invalid token" }, 401);
   401→  }
   402→
   403→  let input: {
   404→    contactId: string;
   405→    categoryId?: string | null;
   406→    transactionType: keyof typeof TRANSACTION_EFFECT;
   407→    amount: number;
   408→    currency: "IRT" | "IRR";
   409→    effectiveAt: string;
   410→    description?: string;
   411→    idempotencyKey: string;
   412→  };
   413→  try {
   414→    input = await req.json();
   415→  } catch {
   416→    return json({ error: "Invalid JSON" }, 400);
   417→  }
   418→
   419→  if (
   420→    !input.contactId ||
   421→    !input.transactionType ||
   422→    !input.amount ||
   423→    !input.currency ||
   424→    !input.effectiveAt ||
   425→    !input.idempotencyKey
   426→  ) {
   427→    return json({ error: "Missing required fields" }, 400);
   428→  }
   429→  if (input.amount <= 0) {
   430→    return json({ error: "Amount must be positive" }, 400);
   431→  }
   432→  if (!TRANSACTION_EFFECT[input.transactionType]) {
   433→    return json({ error: "Invalid transaction type" }, 400);
   434→  }
   435→
   436→  const supabase = createClient(SUPABASE_URL, SERVICE_ROLE_KEY);
   437→
   438→  // ۱. تأیید مالکیت: کاربر مالک دفتر این مخاطب است
   439→  const { data: contact } = await supabase
   440→    .from("contacts")
   441→    .select("id, ledger_id, display_name, ledgers(id, owner_user_id)")
   442→    .eq("id", input.contactId)
   443→    .single();
   444→
   445→  if (!contact) {
   446→    await logAudit(supabase, user.sub, "failure", "Contact not found", "-");
   447→    return json({ error: "Contact not found" }, 404);
   448→  }
   449→  const ledger = (contact.ledgers as Array<{ id: string; owner_user_id: string }>)[0];
   450→  if (!ledger || ledger.owner_user_id !== user.sub) {
   451→    await logAudit(supabase, user.sub, "failure", "Not authorized", "-");
   452→    return json({ error: "Not authorized" }, 403);
   453→  }
   454→
   455→  // ۲. Idempotency: اگر همین تراکنش قبلاً ثبت شده، همان را برگردان
   456→  const { data: existing } = await supabase
   457→    .from("transactions")
   458→    .select("*")
   459→    .eq("ledger_id", ledger.id)
   460→    .eq("idempotency_key", input.idempotencyKey)
   461→    .maybeSingle();
   462→  if (existing) {
   463→    return json({ ok: true, transaction: existing, idempotent: true });
   464→  }
   465→
   466→  // ۳. اعتبارسنجی مالی: دریافت نباید از طلب باز بیشتر باشد
   467→  if (
   468→    input.transactionType === "RECEIPT_ALLOCATION" ||
   469→    input.transactionType === "PAYMENT_ALLOCATION"
   470→  ) {
   471→    const balance = await computeOutstanding(supabase, input.contactId, input.currency);
   472→    if (input.transactionType === "RECEIPT_ALLOCATION" && input.amount > balance.receivable) {
   473→      const msg =
   474→        "Receipt " + input.amount + " exceeds outstanding receivable " + balance.receivable;
   475→      await logAudit(supabase, user.sub, "failure", msg, "-");
   476→      return json({ error: msg }, 400);
   477→    }
   478→    if (input.transactionType === "PAYMENT_ALLOCATION" && input.amount > balance.payable) {
   479→      const msg =
   480→        "Payment " + input.amount + " exceeds outstanding payable " + balance.payable;
   481→      await logAudit(supabase, user.sub, "failure", msg, "-");
   482→      return json({ error: msg }, 400);
   483→    }
   484→  }
   485→
   486→  // ۴. ثبت تراکنش
   487→  const { data: txn, error: insertErr } = await supabase
   488→    .from("transactions")
   489→    .insert({
   490→      ledger_id: ledger.id,
   491→      contact_id: input.contactId,
   492→      category_id: input.categoryId ?? null,
   493→      transaction_type: input.transactionType,
   494→      amount: input.amount,
   495→      currency: input.currency,
   496→      effective_at: input.effectiveAt,
   497→      created_by: user.sub,
   498→      description: input.description ?? null,
   499→      status: "POSTED",
   500→      idempotency_key: input.idempotencyKey,
   501→    })
   502→    .select()
   503→    .single();
   504→
   505→  if (insertErr || !txn) {
   506→    const msg = "Insert failed: " + (insertErr?.message ?? "unknown");
   507→    await logAudit(supabase, user.sub, "failure", msg, "-");
   508→    return json({ error: msg }, 500);
   509→  }
   510→
   511→  // ۵. ممیزی
   512→  await logAudit(
   513→    supabase,
   514→    user.sub,
   515→    "success",
   516→    "Created " + input.transactionType + " " + input.amount + " " + input.currency +
   517→      " for " + contact.display_name,
   518→    txn.id
   519→  );
   520→
   521→  return json({ ok: true, transaction: txn });
   522→});
